$ErrorActionPreference = "Stop"

# -------------------------
# FIXED VARIABLES
# -------------------------
$APP_CLIENT_ID     = $env:APP_REGISTRATION_CLIENT_ID
$APP_CLIENT_SECRET = $env:APP_REGISTRATION_CLIENT_SECRET
$TENANT_ID         = $env:TENANT_ID
$KV_NAME           = $env:akv_name

if ([string]::IsNullOrWhiteSpace($KV_NAME)) {
    throw "KeyVault name not set"
}
Write-Host "Checking if Key Vault '$KV_NAME' exists..."
az keyvault show --name $KV_NAME --only-show-errors | Out-Null
# -------------------------
# LOGIN AS APP REGISTRATION
# -------------------------
Write-Host "Logging in with Service Principal: $APP_CLIENT_ID"

az account clear | Out-Null

az login --service-principal `
  --username $APP_CLIENT_ID `
  --password $APP_CLIENT_SECRET `
  --tenant $TENANT_ID | Out-Null


# -------------------------
# VALIDATIONS
# -------------------------
Write-Host "Starting Key Vault secret upload"


# -------------------------
# VARIABLE → SECRET MAPPING
# -------------------------
$secretMap = @{
    "CATALOG"                     = "CATALOG"
    "DESASCODE"                   = "DESASCODE"
    "JOBCLUSTERDEVR2"             = "JOBCLUSTERDEVR2"
    "LLMENDPOINT"                 = "LLMENDPOINT"
    "MAXITERATIONS"               = "MAXITERATIONS"
    "MAXRETRYATTEMPTS"            = "MAXRETRYATTEMPTS"
    "MLFLOWEXPERIMENTNAME"        = "MLFLOWEXPERIMENTNAME"
    "MLFLOWMODELNAME"             = "MLFLOWMODELNAME"
    "MODELTYPE"                   = "MODELTYPE"
    "RATELIMITCALLSPERMINUTE"     = "RATELIMITCALLSPERMINUTE"
    "REQUESTTIMEOUTSECONDS"       = "REQUESTTIMEOUTSECONDS"
    "SASTOKEN_DE"                 = "SASTOKEN-DE"   # FIXED NAME
    "SCHEMA"                      = "SCHEMA"
    "SLMENDPOINT"                 = "SLMENDPOINT"
    "TEST"                        = "TEST"
}

# -------------------------
# LOOP THROUGH VARIABLES
# -------------------------
foreach ($entry in $secretMap.GetEnumerator()) {

    $varName       = $entry.Key
    $kvSecretName  = $entry.Value

    # Retrieve pipeline variable
    $value = (Get-Item "Env:$varName" -ErrorAction SilentlyContinue).Value

    if ([string]::IsNullOrWhiteSpace($value)) {
        Write-Host " Skipping $varName – value is empty or not set."
        continue
    }

    Write-Host " Setting secret '$kvSecretName' from variable '$varName'"

    az keyvault secret set `
        --vault-name $KV_NAME `
        --name $kvSecretName `
        --value "$value" `
        --only-show-errors -o none | Out-Null

    Write-Host " ✔ Secret uploaded: $kvSecretName"
}

Write-Host "🎉 All secrets updated successfully!"
